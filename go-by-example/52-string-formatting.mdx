<GoExampleRow docs={"Go offers several printing \"verbs\" designed to format general Go values. For example, this prints an instance of our `point` struct."} language="go" code={"    p := point{1, 2}\n    fmt.Printf(\"struct1: %v\\n\", p)"} />

<GoExampleRow docs={"If the value is a struct, the `%+v` variant will include the struct's field names."} language="go" code={"    fmt.Printf(\"struct2: %+v\\n\", p)"} />

<GoExampleRow docs={"The `%#v` variant prints a Go syntax representation of the value, i.e. the source code snippet that would produce that value."} language="go" code={"    fmt.Printf(\"struct3: %#v\\n\", p)"} />

<GoExampleRow docs={"To print the type of a value, use `%T`."} language="go" code={"    fmt.Printf(\"type: %T\\n\", p)"} />

<GoExampleRow docs={"Formatting booleans is straight-forward."} language="go" code={"    fmt.Printf(\"bool: %t\\n\", true)"} />

<GoExampleRow docs={"There are many options for formatting integers. Use `%d` for standard, base-10 formatting."} language="go" code={"    fmt.Printf(\"int: %d\\n\", 123)"} />

<GoExampleRow docs={"This prints a binary representation."} language="go" code={"    fmt.Printf(\"bin: %b\\n\", 14)"} />

<GoExampleRow docs={"This prints the character corresponding to the given integer."} language="go" code={"    fmt.Printf(\"char: %c\\n\", 33)"} />

<GoExampleRow docs={"`%x` provides hex encoding."} language="go" code={"    fmt.Printf(\"hex: %x\\n\", 456)"} />

<GoExampleRow docs={"There are also several formatting options for floats. For basic decimal formatting use `%f`."} language="go" code={"    fmt.Printf(\"float1: %f\\n\", 78.9)"} />

<GoExampleRow docs={"`%e` and `%E` format the float in (slightly different versions of) scientific notation."} language="go" code={"    fmt.Printf(\"float2: %e\\n\", 123400000.0)\n    fmt.Printf(\"float3: %E\\n\", 123400000.0)"} />

<GoExampleRow docs={"For basic string printing use `%s`."} language="go" code={"    fmt.Printf(\"str1: %s\\n\", \"\\\"string\\\"\")"} />

<GoExampleRow docs={"To double-quote strings as in Go source, use `%q`."} language="go" code={"    fmt.Printf(\"str2: %q\\n\", \"\\\"string\\\"\")"} />

<GoExampleRow docs={"As with integers seen earlier, `%x` renders the string in base-16, with two output characters per byte of input."} language="go" code={"    fmt.Printf(\"str3: %x\\n\", \"hex this\")"} />

<GoExampleRow docs={"To print a representation of a pointer, use `%p`."} language="go" code={"    fmt.Printf(\"pointer: %p\\n\", &p)"} />

<GoExampleRow docs={"When formatting numbers you will often want to control the width and precision of the resulting figure. To specify the width of an integer, use a number after the `%` in the verb. By default the result will be right-justified and padded with spaces."} language="go" code={"    fmt.Printf(\"width1: |%6d|%6d|\\n\", 12, 345)"} />

<GoExampleRow docs={"You can also specify the width of printed floats, though usually you'll also want to restrict the decimal precision at the same time with the width.precision syntax."} language="go" code={"    fmt.Printf(\"width2: |%6.2f|%6.2f|\\n\", 1.2, 3.45)"} />

<GoExampleRow docs={"To left-justify, use the `-` flag."} language="go" code={"    fmt.Printf(\"width3: |%-6.2f|%-6.2f|\\n\", 1.2, 3.45)"} />

<GoExampleRow docs={"You may also want to control width when formatting strings, especially to ensure that they align in table-like output. For basic right-justified width."} language="go" code={"    fmt.Printf(\"width4: |%6s|%6s|\\n\", \"foo\", \"b\")"} />

<GoExampleRow docs={"To left-justify use the `-` flag as with numbers."} language="go" code={"    fmt.Printf(\"width5: |%-6s|%-6s|\\n\", \"foo\", \"b\")"} />

<GoExampleRow docs={"So far we've seen `Printf`, which prints the formatted string to `os.Stdout`. `Sprintf` formats and returns a string without printing it anywhere."} language="go" code={"    s := fmt.Sprintf(\"sprintf: a %s\", \"string\")\n    fmt.Println(s)"} />

<GoExampleRow docs={"You can format+print to `io.Writers` other than `os.Stdout` using `Fprintf`."} language="go" code={"    fmt.Fprintf(os.Stderr, \"io: an %s\\n\", \"error\")\n}"} />
