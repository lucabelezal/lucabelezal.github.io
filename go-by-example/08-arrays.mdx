<GoExampleRow docs={"If you specify the index with `:`, the elements in between will be zeroed."} language="go" code={"    b = [...]int{100, 3: 400, 500}\n    fmt.Println(\"idx:\", b)"} />

<GoExampleRow docs={"Array types are one-dimensional, but you can compose types to build multi-dimensional data structures."} language="go" code={"    var twoD [2][3]int\n    for i := range 2 {\n        for j := range 3 {\n            twoD[i][j] = i + j\n        }\n    }\n    fmt.Println(\"2d: \", twoD)"} />

<GoExampleRow docs={"You can create and initialize multi-dimensional arrays at once too."} language="go" code={"    twoD = [2][3]int{\n        {1, 2, 3},\n        {1, 2, 3},\n    }\n    fmt.Println(\"2d: \", twoD)\n}"} />
