<GoExampleRow docs={"We'll start with a simple command that takes no arguments or input and just prints something to stdout. The `exec.Command` helper creates an object to represent this external process."} language="go" code={"    dateCmd := exec.Command(\"date\")"} />

<GoExampleRow docs={"The `Output` method runs the command, waits for it to finish and collects its standard output.  If there were no errors, `dateOut` will hold bytes with the date info."} language="go" code={"    dateOut, err := dateCmd.Output()\n    if err != nil {\n        panic(err)\n    }\n    fmt.Println(\"> date\")\n    fmt.Println(string(dateOut))"} />

<GoExampleRow docs={"`Output` and other methods of `Command` will return `*exec.Error` if there was a problem executing the command (e.g. wrong path), and `*exec.ExitError` if the command ran but exited with a non-zero return code."} language="go" code={"    _, err = exec.Command(\"date\", \"-x\").Output()\n    if err != nil {\n        if e, ok := errors.AsType[*exec.Error](err); ok {\n            fmt.Println(\"failed executing:\", e)\n        } else if e, ok := errors.AsType[*exec.ExitError](err); ok {\n            exitCode := e.ExitCode()\n            fmt.Println(\"command exit rc =\", exitCode)\n        } else {\n            panic(err)\n        }\n    }"} />

<GoExampleRow docs={"Next we'll look at a slightly more involved case where we pipe data to the external process on its `stdin` and collect the results from its `stdout`."} language="go" code={"    grepCmd := exec.Command(\"grep\", \"hello\")"} />

<GoExampleRow docs={"Here we explicitly grab input/output pipes, start the process, write some input to it, read the resulting output, and finally wait for the process to exit."} language="go" code={"    grepIn, _ := grepCmd.StdinPipe()\n    grepOut, _ := grepCmd.StdoutPipe()\n    grepCmd.Start()\n    grepIn.Write([]byte(\"hello grep\\ngoodbye grep\"))\n    grepIn.Close()\n    grepBytes, _ := io.ReadAll(grepOut)\n    grepCmd.Wait()"} />

<GoExampleRow docs={"We omitted error checks in the above example, but you could use the usual `if err != nil` pattern for all of them. We also only collect the `StdoutPipe` results, but you could collect the `StderrPipe` in exactly the same way."} language="go" code={"    fmt.Println(\"> grep hello\")\n    fmt.Println(string(grepBytes))"} />

<GoExampleRow docs={"Note that when spawning commands we need to provide an explicitly delineated command and argument array, vs. being able to just pass in one command-line string. If you want to spawn a full command with a string, you can use `bash`'s `-c` option:"} language="go" code={"    lsCmd := exec.Command(\"bash\", \"-c\", \"ls -a -l -h\")\n    lsOut, err := lsCmd.Output()\n    if err != nil {\n        panic(err)\n    }\n    fmt.Println(\"> ls -a -l -h\")\n    fmt.Println(string(lsOut))\n}"} />
