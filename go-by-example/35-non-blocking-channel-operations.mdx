import "fmt"

func main() {
    messages := make(chan string)
    signals := make(chan bool)
    select {
    case msg := <-messages:
        fmt.Println("received message", msg)
    default:
        fmt.Println("no message received")
    }
    msg := "hi"
    select {
    case messages <- msg:
        fmt.Println("sent message", msg)
    default:
        fmt.Println("no message sent")
    }
    select {
    case msg := <-messages:
        fmt.Println("received message", msg)
    case sig := <-signals:
        fmt.Println("received signal", sig)
    default:
        fmt.Println("no activity")
    }
}`}
  sections={[
    {text: "Um receive não-bloqueante. Se houver valor em `messages`, o `select` pega o `case <-messages`. Caso contrário, cai no `default` imediatamente."},

    {text: "Podemos usar múltiplos `case`s acima do `default` para um select não-bloqueante multi-way. Aqui tentamos receives não-bloqueantes em `messages` e `signals`."},
