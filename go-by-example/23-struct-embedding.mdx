  code={`package main

import "fmt"

type base struct {
    num int
}

func (b base) describe() string {
    return fmt.Sprintf("base with num=%v", b.num)
}

type container struct {
    base
    str string
}

func main() {
    co := container{
        base: base{
            num: 1,
        },
        str: "some name",
    }
    fmt.Printf("co={num: %v, str: %v}\n", co.num, co.str)
    fmt.Println("also num:", co.base.num)
    fmt.Println("describe:", co.describe())
    type describer interface {
        describe() string
    }
    var d describer = co
    fmt.Println("describer:", d.describe())
}`}
  sections={[
    {text: "`container` faz _embed_ de `base`. Parece um campo sem nome."},
    {text: "Ao criar structs com literais, precisamos inicializar o embedding explicitamente; o tipo embutido serve como nome do campo."},
    {text: "Podemos acessar campos de `base` diretamente em `co`, ex. `co.num`."},
    {text: "Alternativamente, pode usar o caminho completo com o nome do tipo embutido."},
    {text: "Como `container` embute `base`, os métodos de `base` viram métodos de `container`. Aqui invocamos um método embutido diretamente em `co`."},
    {text: "Embedding com métodos pode dar implementações de interface a outros structs. Aqui `container` implementa `describer` porque embute `base`."},
