<GoExampleRow docs={"Perhaps the most basic file reading task is slurping a file's entire contents into memory."} language="go" code={"    path := filepath.Join(os.TempDir(), \"dat\")\n    dat, err := os.ReadFile(path)\n    check(err)\n    fmt.Print(string(dat))"} />

<GoExampleRow docs={"You'll often want more control over how and what parts of a file are read. For these tasks, start by `Open`ing a file to obtain an `os.File` value."} language="go" code={"    f, err := os.Open(path)\n    check(err)"} />

<GoExampleRow docs={"Read some bytes from the beginning of the file. Allow up to 5 to be read but also note how many actually were read."} language="go" code={"    b1 := make([]byte, 5)\n    n1, err := f.Read(b1)\n    check(err)\n    fmt.Printf(\"%d bytes: %s\\n\", n1, string(b1[:n1]))"} />

<GoExampleRow docs={"You can also `Seek` to a known location in the file and `Read` from there."} language="go" code={"    o2, err := f.Seek(6, io.SeekStart)\n    check(err)\n    b2 := make([]byte, 2)\n    n2, err := f.Read(b2)\n    check(err)\n    fmt.Printf(\"%d bytes @ %d: \", n2, o2)\n    fmt.Printf(\"%v\\n\", string(b2[:n2]))"} />

<GoExampleRow docs={"Other methods of seeking are relative to the current cursor position,"} language="go" code={"    _, err = f.Seek(2, io.SeekCurrent)\n    check(err)"} />

<GoExampleRow docs={"and relative to the end of the file."} language="go" code={"    _, err = f.Seek(-4, io.SeekEnd)\n    check(err)"} />

<GoExampleRow docs={"The `io` package provides some functions that may be helpful for file reading. For example, reads like the ones above can be more robustly implemented with `ReadAtLeast`."} language="go" code={"    o3, err := f.Seek(6, io.SeekStart)\n    check(err)\n    b3 := make([]byte, 2)\n    n3, err := io.ReadAtLeast(f, b3, 2)\n    check(err)\n    fmt.Printf(\"%d bytes @ %d: %s\\n\", n3, o3, string(b3))"} />

<GoExampleRow docs={"There is no built-in rewind, but `Seek(0, io.SeekStart)` accomplishes this."} language="go" code={"    _, err = f.Seek(0, io.SeekStart)\n    check(err)"} />

<GoExampleRow docs={"The `bufio` package implements a buffered reader that may be useful both for its efficiency with many small reads and because of the additional reading methods it provides."} language="go" code={"    r4 := bufio.NewReader(f)\n    b4, err := r4.Peek(5)\n    check(err)\n    fmt.Printf(\"5 bytes: %s\\n\", string(b4))"} />

<GoExampleRow docs={"Close the file when you're done (usually this would be scheduled immediately after `Open`ing with `defer`)."} language="go" code={"    f.Close()\n}"} />
