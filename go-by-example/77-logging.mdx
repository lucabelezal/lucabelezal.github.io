    mylog.Println("from mylog")

    mylog.Println("from mylog")
    var buf bytes.Buffer
    buflog := log.New(&buf, "buf:", log.LstdFlags)
    buflog.Println("hello")
    fmt.Print("from buflog:", buf.String())

    myslog := slog.New(jsonHandler)
    myslog.Info("hi there")

}`}
