<GoExampleRow docs={"Simply invoking functions like `Println` from the `log` package uses the _standard_ logger, which is already pre-configured for reasonable logging output to `os.Stderr`. Additional methods like `Fatal*` or `Panic*` will exit the program after logging."} language="go" code={"    log.Println(\"standard logger\")"} />

<GoExampleRow docs={"Loggers can be configured with _flags_ to set their output format. By default, the standard logger has the `log.Ldate` and `log.Ltime` flags set, and these are collected in `log.LstdFlags`. We can change its flags to emit time with microsecond accuracy, for example."} language="go" code={"    log.SetFlags(log.LstdFlags | log.Lmicroseconds)\n    log.Println(\"with micro\")"} />

<GoExampleRow docs={"It also supports emitting the file name and line from which the `log` function is called."} language="go" code={"    log.SetFlags(log.LstdFlags | log.Lshortfile)\n    log.Println(\"with file/line\")"} />

<GoExampleRow docs={"It may be useful to create a custom logger and pass it around. When creating a new logger, we can set a _prefix_ to distinguish its output from other loggers."} language="go" code={"    mylog := log.New(os.Stdout, \"my:\", log.LstdFlags)\n    mylog.Println(\"from mylog\")"} />

<GoExampleRow docs={"We can set the prefix on existing loggers (including the standard one) with the `SetPrefix` method."} language="go" code={"    mylog.SetPrefix(\"ohmy:\")\n    mylog.Println(\"from mylog\")"} />

<GoExampleRow docs={"Loggers can have custom output targets; any `io.Writer` works."} language="go" code={"    var buf bytes.Buffer\n    buflog := log.New(&buf, \"buf:\", log.LstdFlags)"} />

<GoExampleRow docs={"This call writes the log output into `buf`."} language="go" code={"    buflog.Println(\"hello\")"} />

<GoExampleRow docs={"This will actually show it on standard output."} language="go" code={"    fmt.Print(\"from buflog:\", buf.String())"} />

<GoExampleRow docs={"The `slog` package provides _structured_ log output. For example, logging in JSON format is straightforward."} language="go" code={"    jsonHandler := slog.NewJSONHandler(os.Stderr, nil)\n    myslog := slog.New(jsonHandler)\n    myslog.Info(\"hi there\")"} />

<GoExampleRow docs={"In addition to the message, `slog` output can contain an arbitrary number of key=value pairs."} language="go" code={"    myslog.Info(\"hello again\", \"key\", \"val\", \"age\", 25)\n}"} />
