<GoExampleRow docs={"First we'll look at basic rate limiting. Suppose we want to limit our handling of incoming requests. We'll serve these requests off a channel of the same name."} language="go" code={"    requests := make(chan int, 5)\n    for i := 1; i <= 5; i++ {\n        requests <- i\n    }\n    close(requests)"} />

<GoExampleRow docs={"This `limiter` channel will receive a value every 200 milliseconds. This is the regulator in our rate limiting scheme."} language="go" code={"    limiter := time.Tick(200 * time.Millisecond)"} />

<GoExampleRow docs={"By blocking on a receive from the `limiter` channel before serving each request, we limit ourselves to 1 request every 200 milliseconds."} language="go" code={"    for req := range requests {\n        <-limiter\n        fmt.Println(\"request\", req, time.Now())\n    }"} />

<GoExampleRow docs={"We may want to allow short bursts of requests in our rate limiting scheme while preserving the overall rate limit. We can accomplish this by buffering our limiter channel. This `burstyLimiter` channel will allow bursts of up to 3 events."} language="go" code={"    burstyLimiter := make(chan time.Time, 3)"} />

<GoExampleRow docs={"Fill up the channel to represent allowed bursting."} language="go" code={"    for range 3 {\n        burstyLimiter <- time.Now()\n    }"} />

<GoExampleRow docs={"Every 200 milliseconds we'll try to add a new value to `burstyLimiter`, up to its limit of 3."} language="go" code={"    go func() {\n        for t := range time.Tick(200 * time.Millisecond) {\n            burstyLimiter <- t\n        }\n    }()"} />

<GoExampleRow docs={"Now simulate 5 more incoming requests. The first 3 of these will benefit from the burst capability of `burstyLimiter`."} language="go" code={"    burstyRequests := make(chan int, 5)\n    for i := 1; i <= 5; i++ {\n        burstyRequests <- i\n    }\n    close(burstyRequests)\n    for req := range burstyRequests {\n        <-burstyLimiter\n        fmt.Println(\"request\", req, time.Now())\n    }\n}"} />
