<GoExampleRow docs={"This tests whether a pattern matches a string."} language="go" code={"    match, _ := regexp.MatchString(\"p([a-z]+)ch\", \"peach\")\n    fmt.Println(match)"} />

<GoExampleRow docs={"Above we used a string pattern directly, but for other regexp tasks you'll need to `Compile` an optimized `Regexp` struct."} language="go" code={"    r, _ := regexp.Compile(\"p([a-z]+)ch\")"} />

<GoExampleRow docs={"Many methods are available on these structs. Here's a match test like we saw earlier."} language="go" code={"    fmt.Println(r.MatchString(\"peach\"))"} />

<GoExampleRow docs={"This finds the match for the regexp."} language="go" code={"    fmt.Println(r.FindString(\"peach punch\"))"} />

<GoExampleRow docs={"This also finds the first match but returns the start and end indexes for the match instead of the matching text."} language="go" code={"    fmt.Println(\"idx:\", r.FindStringIndex(\"peach punch\"))"} />

<GoExampleRow docs={"The `Submatch` variants include information about both the whole-pattern matches and the submatches within those matches. For example this will return information for both `p([a-z]+)ch` and `([a-z]+)`."} language="go" code={"    fmt.Println(r.FindStringSubmatch(\"peach punch\"))"} />

<GoExampleRow docs={"Similarly this will return information about the indexes of matches and submatches."} language="go" code={"    fmt.Println(r.FindStringSubmatchIndex(\"peach punch\"))"} />

<GoExampleRow docs={"The `All` variants of these functions apply to all matches in the input, not just the first. For example to find all matches for a regexp."} language="go" code={"    fmt.Println(r.FindAllString(\"peach punch pinch\", -1))"} />

<GoExampleRow docs={"These `All` variants are available for the other functions we saw above as well."} language="go" code={"    fmt.Println(\"all:\", r.FindAllStringSubmatchIndex(\n        \"peach punch pinch\", -1))"} />

<GoExampleRow docs={"Providing a non-negative integer as the second argument to these functions will limit the number of matches."} language="go" code={"    fmt.Println(r.FindAllString(\"peach punch pinch\", 2))"} />

<GoExampleRow docs={"Our examples above had string arguments and used names like `MatchString`. We can also provide `[]byte` arguments and drop `String` from the function name."} language="go" code={"    fmt.Println(r.Match([]byte(\"peach\")))"} />

<GoExampleRow docs={"When creating global variables with regular expressions you can use the `MustCompile` variation of `Compile`. `MustCompile` panics instead of returning an error, which makes it safer to use for global variables."} language="go" code={"    r = regexp.MustCompile(\"p([a-z]+)ch\")\n    fmt.Println(\"regexp:\", r)"} />

<GoExampleRow docs={"The `regexp` package can also be used to replace subsets of strings with other values."} language="go" code={"    fmt.Println(r.ReplaceAllString(\"a peach\", \"<fruit>\"))"} />

<GoExampleRow docs={"The `Func` variant allows you to transform matched text with a given function."} language="go" code={"    in := []byte(\"a peach\")\n    out := r.ReplaceAllFunc(in, bytes.ToUpper)\n    fmt.Println(string(out))\n}"} />
