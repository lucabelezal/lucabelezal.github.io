<GoExampleRow docs={"In this example our state will be owned by a single goroutine. This will guarantee that the data is never corrupted with concurrent access. In order to read or write that state, other goroutines will send messages to the owning goroutine and receive corresponding replies. These `readOp` and `writeOp` `struct`s encapsulate those requests and a way for the owning goroutine to respond."} language="go" code={"type readOp struct {\n    key  int\n    resp chan int\n}\ntype writeOp struct {\n    key  int\n    val  int\n    resp chan bool\n}"} />

<GoExampleRow docs={undefined} language="go" code={"func main() {"} />

<GoExampleRow docs={"As before we'll count how many operations we perform."} language="go" code={"    var readOps uint64\n    var writeOps uint64"} />

<GoExampleRow docs={"The `reads` and `writes` channels will be used by other goroutines to issue read and write requests, respectively."} language="go" code={"    reads := make(chan readOp)\n    writes := make(chan writeOp)"} />

<GoExampleRow docs={"Here is the goroutine that owns the `state`, which is a map as in the previous example but now private to the stateful goroutine. This goroutine repeatedly selects on the `reads` and `writes` channels, responding to requests as they arrive. A response is executed by first performing the requested operation and then sending a value on the response channel `resp` to indicate success (and the desired value in the case of `reads`)."} language="go" code={"    go func() {\n        var state = make(map[int]int)\n        for {\n            select {\n            case read := <-reads:\n                read.resp <- state[read.key]\n            case write := <-writes:\n                state[write.key] = write.val\n                write.resp <- true\n            }\n        }\n    }()"} />

<GoExampleRow docs={"This starts 100 goroutines to issue reads to the state-owning goroutine via the `reads` channel. Each read requires constructing a `readOp`, sending it over the `reads` channel, and then receiving the result over the provided `resp` channel."} language="go" code={"    for range 100 {\n        go func() {\n            for {\n                read := readOp{\n                    key:  rand.Intn(5),\n                    resp: make(chan int)}\n                reads <- read\n                <-read.resp\n                atomic.AddUint64(&readOps, 1)\n                time.Sleep(time.Millisecond)\n            }\n        }()\n    }"} />

<GoExampleRow docs={"We start 10 writes as well, using a similar approach."} language="go" code={"    for range 10 {\n        go func() {\n            for {\n                write := writeOp{\n                    key:  rand.Intn(5),\n                    val:  rand.Intn(100),\n                    resp: make(chan bool)}\n                writes <- write\n                <-write.resp\n                atomic.AddUint64(&writeOps, 1)\n                time.Sleep(time.Millisecond)\n            }\n        }()\n    }"} />

<GoExampleRow docs={"Let the goroutines work for a second."} language="go" code={"    time.Sleep(time.Second)"} />

<GoExampleRow docs={"Finally, capture and report the op counts."} language="go" code={"    readOpsFinal := atomic.LoadUint64(&readOps)\n    fmt.Println(\"readOps:\", readOpsFinal)\n    writeOpsFinal := atomic.LoadUint64(&writeOps)\n    fmt.Println(\"writeOps:\", writeOpsFinal)\n}"} />
