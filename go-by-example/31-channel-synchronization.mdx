import (
    "fmt"
    "time"
)

func worker(done chan bool) {
    fmt.Print("working...")
    time.Sleep(time.Second)
    fmt.Println("done")
    done <- true
}
func main() {
    done := make(chan bool, 1)
    go worker(done)
    <-done
}`}
  sections={[

    {text: "Inicie uma goroutine worker, passando o channel para notificação."},
    {text: "Bloqueie até receber notificação do worker no channel."},
