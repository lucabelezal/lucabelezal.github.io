import (
    "fmt"
    "time"
)
func worker(done chan bool) {
    fmt.Print("working...")
    time.Sleep(time.Second)
    fmt.Println("done")
    done <- true
}
func main() {
    done := make(chan bool, 1)
    go worker(done)
    <-done
}`}

    {text: "Start a worker goroutine, giving it the channel to notify on."},
    {text: "Block until we receive a notification from the worker on the channel."},
