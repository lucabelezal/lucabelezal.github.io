<GoExampleRow docs={"`Format` and `Parse` use example-based layouts. Usually you'll use a constant from `time` for these layouts, but you can also supply custom layouts. Layouts must use the reference time `Mon Jan 2 15:04:05 MST 2006` to show the pattern with which to format/parse a given time/string. The example time must be exactly as shown: the year 2006, 15 for the hour, Monday for the day of the week, etc."} language="go" code={"    p(t.Format(\"3:04PM\"))\n    p(t.Format(\"Mon Jan _2 15:04:05 2006\"))\n    p(t.Format(\"2006-01-02T15:04:05.999999-07:00\"))\n    form := \"3 04 PM\"\n    t2, _ := time.Parse(form, \"8 41 PM\")\n    p(t2)"} />

<GoExampleRow docs={"For purely numeric representations you can also use standard string formatting with the extracted components of the time value."} language="go" code={"    fmt.Printf(\"%d-%02d-%02dT%02d:%02d:%02d-00:00\\n\",\n        t.Year(), t.Month(), t.Day(),\n        t.Hour(), t.Minute(), t.Second())"} />
